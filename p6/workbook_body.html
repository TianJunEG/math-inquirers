Optional, for children who already code: Python version. % is mod. Change the message and shift to test Q2; a negative shift decodes.
alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
message = "MEET AT THE MRT"
shift = 3
code = ""
for letter in message:
    if letter in alphabet:
        n = alphabet.index(letter)   # A is 0
        m = (n + shift) % 26         # % is mod
        code = code + alphabet[m]
    else:
        code = code + letter         # keep spaces
print(code)
